/*
 * QUESTION 4 (DAA) : Solve the 0-1 Knapsack problem using dynamic programming.
 *
 * Given n items (each with a value and a weight) and a knapsack of integer
 * capacity W, choose a SUBSET of items (you can either take an item fully or
 * not at all - no fractions) so that the total value is maximum and the total
 * weight does not exceed W.
 *
 * HOW IT WORKS (TOP-DOWN DYNAMIC PROGRAMMING):
 *   1. Define solve(i, w) as the maximum value achievable using the first i
 *      items and a knapsack capacity of w.
 *   2. Recursively consider skipping item i or taking it when it fits, then
 *      memoize the result so each (i, w) state is solved only once.
 *   3. solve(n, W) is the answer. The memo table is also used to trace the
 *      chosen items backwards.
 *
 * COMPLEXITY:
 *   - Time  : O(n * W)  (pseudo-polynomial; each state is solved once)
 *   - Space : O(n * W)  (memo table plus O(n) recursion depth)
 *
 * NOTES / INFO:
 *   - Greedy does NOT work here (a high-ratio item may block several small
 *     items whose total value is larger), which is why DP is required.
 *   - Weights must be integers because they index the table columns.
 *   - Unlike the plain DP, this version also prints the chosen item numbers.
  *
 * COMMANDS (run from inside this folder):
 *   install dependencies : sudo apt install g++
 *   compile and run      : g++ -std=c++17 -Wall q4_Knapsack01_DP.cpp -o q4_Knapsack01_DP && ./q4_Knapsack01_DP
*/


#include <algorithm>
#include <iostream>
#include <vector>

using namespace std;

// Function to read the value and weight of each item
void readItems(int itemCount, vector<long long>& value,
               vector<int>& weight) {

    cout << "Enter value and weight of each item:\n";

    for (int i = 0; i < itemCount; ++i) {

        // Read the value of the current item
        cout << "Item " << (i + 1) << " value: ";
        cin >> value[i];

        // Read the weight of the current item
        cout << "Item " << (i + 1) << " weight: ";
        cin >> weight[i];
    }
}

// Function to find the maximum value using recursion and memoization
long long solveKnapsack(const vector<long long>& value,
                        const vector<int>& weight,
                        vector<vector<long long>>& memo,
                        int itemCount, int capacity) {

    // Base condition: no items left, so maximum value is 0
    if (itemCount == 0)
        return 0;

    // If this state is already calculated, return stored result
    if (memo[itemCount][capacity] != -1)
        return memo[itemCount][capacity];

    // Option 1: Do not select the current item
    long long best = solveKnapsack(value, weight, memo,
                                   itemCount - 1, capacity);

    // Option 2: Select the current item if its weight fits
    if (weight[itemCount - 1] <= capacity) {

        long long withItem = value[itemCount - 1] +
                             solveKnapsack(value, weight, memo,
                                           itemCount - 1,
                                           capacity - weight[itemCount - 1]);

        // Choose the option that gives maximum value
        best = max(best, withItem);
    }

    // Store the calculated answer in the memoization table
    memo[itemCount][capacity] = best;

    return best;
}

// Function to print which items are selected
void printSelectedItems(const vector<long long>& value,
                        const vector<int>& weight,
                        vector<vector<long long>>& memo,
                        int itemCount, int capacity) {

    cout << "Selected items:\n";

    int w = capacity;

    // Trace the solution backwards through the items
    for (int i = itemCount; i > 0; --i) {

        // Maximum value when considering the current item
        long long withItems = solveKnapsack(value, weight, memo, i, w);

        // Maximum value when the current item is not considered
        long long withoutItem = solveKnapsack(value, weight, memo,
                                              i - 1, w);

        // If the values differ, the current item was selected
        if (withItems != withoutItem) {

            cout << "Item " << i
                 << " (value=" << value[i - 1]
                 << ", weight=" << weight[i - 1] << ")\n";

            // Reduce remaining capacity after selecting the item
            w -= weight[i - 1];
        }
    }
}

// Main function to execute the program
void run() {

    int itemCount;

    // Read the total number of items
    cout << "Enter number of items: ";
    cin >> itemCount;

    // Create arrays to store values and weights
    vector<long long> value(itemCount);
    vector<int> weight(itemCount);

    // Read item details
    readItems(itemCount, value, weight);

    int capacity;

    // Read the maximum capacity of the knapsack
    cout << "Enter knapsack capacity: ";
    cin >> capacity;

    // Create memoization table initialized with -1
    vector<vector<long long>> memo(
        itemCount + 1,
        vector<long long>(capacity + 1, -1)
    );

    // Calculate the maximum possible value
    long long maxValue = solveKnapsack(
        value, weight, memo, itemCount, capacity
    );

    // Display the maximum value
    cout << "\nMaximum value = " << maxValue << "\n";

    // Display the selected items
    printSelectedItems(value, weight, memo, itemCount, capacity);
}

// Program execution starts here
int main() {
    run();
    return 0;
}
