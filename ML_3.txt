
"""

Bank Customer Churn Prediction using Artificial Neural Network.

Objective:
Predict whether a bank customer will leave the bank (Churn = 1)
or stay with the bank (Churn = 0).

Steps:
1. Import libraries.
2. Load dataset.
3. Separate features and target.
4. Split data into training and testing sets.
5. Normalize the data.
6. Build and train the baseline neural network.
7. Build and train the improved neural network.
8. Evaluate both models using accuracy and confusion matrix.
9. Display graphs.
"""

# %% STEP 1: IMPORT LIBRARIES

import os
import pandas as pd
import matplotlib.pyplot as plt

# Import functions for splitting data and evaluation
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import accuracy_score, confusion_matrix

# Import TensorFlow Keras components for building neural network
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Dense, Dropout, Input
from tensorflow.keras.callbacks import EarlyStopping


# Find the folder where the Python file is located
if "__file__" in globals():
    SCRIPT_DIR = os.path.dirname(os.path.abspath(__file__))
else:
    SCRIPT_DIR = os.getcwd()

# Dataset file name
DATASET_FILE = os.path.join(SCRIPT_DIR, "Churn_Modelling.csv")


# %% STEP 2: CREATE DEMO DATASET IF FILE IS MISSING

# If the original Kaggle dataset is missing,
# create a small demo dataset so the program can run.

# Check whether the original dataset exists
if os.path.exists(DATASET_FILE):
    print("Using original dataset:", DATASET_FILE)

else:
    # Create demo CSV file if dataset is unavailable
    with open(DATASET_FILE, "w", encoding="utf-8") as file:
        file.write(DEMO_CHURN_CSV)

    print("Created demo dataset:", DATASET_FILE)


# %% STEP 3: LOAD THE DATASET

# Read CSV file using Pandas
data = pd.read_csv(DATASET_FILE)

# Remove unnecessary columns because they do not help
# predict whether a customer will leave.
data = data.drop(columns=["RowNumber", "CustomerId", "Surname"])

# Display dataset information
print("\nDataset Shape:", data.shape)
print(data.head())


# %% STEP 4: SEPARATE FEATURES AND TARGET

# Convert categorical columns into numerical columns.
# Geography and Gender contain text values.
# get_dummies() converts them into 0 and 1.
data = pd.get_dummies(
    data,
    columns=["Geography", "Gender"],
    drop_first=True
)

# X contains input features used for prediction
X = data.drop(columns=["Exited"])

# y contains the target/output column
# 0 = Customer stays
# 1 = Customer leaves
y = data["Exited"]

# Divide dataset into training (80%) and testing (20%)
# random_state=42 gives reproducible results.
# stratify=y maintains the class proportions.
X_train, X_test, y_train, y_test = train_test_split(
    X,
    y,
    test_size=0.20,
    random_state=42,
    stratify=y
)

print("\nTraining Data:", X_train.shape)
print("Testing Data :", X_test.shape)


# %% STEP 5: NORMALIZE THE DATA

# StandardScaler converts features to a common scale.
scaler = StandardScaler()

# Fit scaler on training data and transform it
X_train = scaler.fit_transform(X_train)

# Apply the same scaling to testing data
# Never fit the scaler separately on test data.
X_test = scaler.transform(X_test)

print("\nData Normalization Completed")


# %% STEP 6: BUILD BASELINE NEURAL NETWORK

# Sequential creates a neural network layer by layer.
initial_model = Sequential([

    # Input layer receives all input features
    Input(shape=(X_train.shape[1],)),

    # First hidden layer with 64 neurons
    Dense(64, activation="relu"),

    # Second hidden layer with 32 neurons
    Dense(32, activation="relu"),

    # Output layer with one neuron
    # Sigmoid produces probability between 0 and 1
    Dense(1, activation="sigmoid")
])

# Compile the neural network
initial_model.compile(
    optimizer="adam",
    loss="binary_crossentropy",
    metrics=["accuracy"]
)

# Display model architecture
initial_model.summary()


# %% STEP 7: TRAIN BASELINE MODEL

# Train the baseline neural network
initial_history = initial_model.fit(
    X_train,
    y_train,

    # Number of training iterations
    epochs=50,

    # Number of samples processed at once
    batch_size=32,

    # Use 20% of training data for validation
    validation_split=0.2,

    verbose=1
)


# %% STEP 8: BUILD IMPROVED NEURAL NETWORK

# Improved model adds:
# 1. Extra hidden layer.
# 2. Dropout to reduce overfitting.
# 3. Early Stopping during training.

model = Sequential([

    # Input layer
    Input(shape=(X_train.shape[1],)),

    # First hidden layer
    Dense(64, activation="relu"),

    # Dropout randomly disables 30% neurons during training
    Dropout(0.3),

    # Second hidden layer
    Dense(32, activation="relu"),

    # Dropout randomly disables 20% neurons during training
    Dropout(0.2),

    # Additional hidden layer
    Dense(16, activation="relu"),

    # Output layer for binary classification
    Dense(1, activation="sigmoid")
])

# Compile improved model
model.compile(
    optimizer="adam",
    loss="binary_crossentropy",
    metrics=["accuracy"]
)

# Display improved model structure
model.summary()


# %% STEP 9: TRAIN IMPROVED MODEL

# EarlyStopping monitors validation loss.
# If validation loss does not improve for 5 epochs,
# training stops automatically.
# restore_best_weights restores the best model weights.

early_stop = EarlyStopping(
    monitor="val_loss",
    patience=5,
    restore_best_weights=True
)

# Train improved neural network
history = model.fit(
    X_train,
    y_train,
    epochs=50,
    batch_size=32,
    validation_split=0.2,
    callbacks=[early_stop],
    verbose=1
)


# %% STEP 10: EVALUATE BOTH MODELS

# Function to calculate accuracy and confusion matrix
def evaluate(name, prediction):

    print("\n-----", name, "-----")

    # Calculate accuracy of the model
    print(
        "Accuracy Score:",
        accuracy_score(y_test, prediction)
    )

    # Display confusion matrix
    # TN = Correctly predicted customers who stay
    # FP = Customers predicted to leave but actually stay
    # FN = Customers predicted to stay but actually leave
    # TP = Correctly predicted customers who leave

    print("\nConfusion Matrix:")
    print(confusion_matrix(y_test, prediction))


# Predict using baseline model
# Probability greater than 0.5 means customer leaves.
y_base = (
    initial_model.predict(X_test) > 0.5
).astype(int).ravel()

# Evaluate baseline model
evaluate("BASELINE MODEL", y_base)


# Predict using improved model
y_pred = (
    model.predict(X_test) > 0.5
).astype(int).ravel()

# Evaluate improved model
evaluate("IMPROVED MODEL", y_pred)


# %% STEP 11: OBSERVATION

print("\nOBSERVATION")

# Compare accuracy of both models
base_accuracy = accuracy_score(y_test, y_base)
improved_accuracy = accuracy_score(y_test, y_pred)

print("Baseline Accuracy :", round(base_accuracy, 4))
print("Improved Accuracy :", round(improved_accuracy, 4))

if improved_accuracy > base_accuracy:
    print("Improved model has higher accuracy.")

elif improved_accuracy < base_accuracy:
    print("Baseline model has higher accuracy.")

else:
    print("Both models have equal accuracy.")

print("Dropout and Early Stopping help reduce overfitting.")


# %% STEP 12: PLOT TRAINING AND VALIDATION GRAPHS

# Plot accuracy and loss for both models
for metric in ["accuracy", "loss"]:

    fig, axis = plt.subplots(figsize=(7, 5))

    # Baseline training and validation graphs
    axis.plot(
        initial_history.history[metric],
        label="Baseline Training"
    )

    axis.plot(
        initial_history.history[f"val_{metric}"],
        "--",
        label="Baseline Validation"
    )

    # Improved training and validation graphs
    axis.plot(
        history.history[metric],
        label="Improved Training"
    )

    axis.plot(
        history.history[f"val_{metric}"],
        "--",
        label="Improved Validation"
    )

    axis.set_title(f"Model {metric}")
    axis.set_xlabel("Epoch")
    axis.set_ylabel(metric.capitalize())
    axis.legend()
    axis.grid(alpha=0.3)

    plt.tight_layout()
    plt.show()


# %% STEP 13: PLOT CONFUSION MATRICES

# Display confusion matrices for both models
for name, prediction in [
    ("Baseline", y_base),
    ("Improved", y_pred)
]:

    # Calculate confusion matrix
    matrix = confusion_matrix(y_test, prediction)

    fig, axis = plt.subplots(figsize=(6, 5))

    # Display confusion matrix as heatmap
    image = axis.imshow(matrix, cmap="Blues")

    # Display numerical values inside the matrix
    for row in range(matrix.shape[0]):
        for column in range(matrix.shape[1]):

            axis.text(
                column,
                row,
                matrix[row, column],
                ha="center",
                va="center"
            )

    axis.set_title(f"{name} Model Confusion Matrix")

    axis.set_xlabel("Predicted (0 = Stay, 1 = Churn)")
    axis.set_ylabel("Actual (0 = Stay, 1 = Churn)")

    axis.set_xticks([0, 1])
    axis.set_yticks([0, 1])

    fig.colorbar(image, ax=axis)
    plt.tight_layout()
    plt.show()
